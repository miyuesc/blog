// 性能监控工具类
class VuePerformanceMonitor {
  private metrics: Map<string, number[]> = new Map()
  
  // 首次内容绘制 (FCP)
  measureFCP() {
    new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        if (entry.name === 'first-contentful-paint') {
          this.recordMetric('FCP', entry.startTime)
          console.log(`🎨 FCP: ${entry.startTime.toFixed(2)}ms`)
        }
      }
    }).observe({ entryTypes: ['paint'] })
  }
  
  // 最大内容绘制 (LCP)
  measureLCP() {
    new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        if (entry.startTime < 10000) { // 只关注前10秒
          this.recordMetric('LCP', entry.startTime)
          console.log(`🖼️ LCP: ${entry.startTime.toFixed(2)}ms`)
        }
      }
    }).observe({ entryTypes: ['largest-contentful-paint'] })
  }
  
  // 首次输入延迟 (FID)
  measureFID() {
    new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        this.recordMetric('FID', entry.processingStart - entry.startTime)
        console.log(`👆 FID: ${(entry.processingStart - entry.startTime).toFixed(2)}ms`)
      }
    }).observe({ entryTypes: ['first-input'] })
  }
  
  // 累积布局偏移 (CLS)
  measureCLS() {
    let clsValue = 0
    new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        if (!entry.hadRecentInput) {
          clsValue += entry.value
          this.recordMetric('CLS', clsValue)
          console.log(`📐 CLS: ${clsValue.toFixed(4)}`)
        }
      }
    }).observe({ entryTypes: ['layout-shift'] })
  }
  
  // Vue特定指标
  measureVueMetrics() {
    // 组件挂载时间
    const originalMount = Vue.component.__vccOpts?.mounted
    Vue.component.__vccOpts.mounted = function() {
      const start = performance.now()
      if (originalMount) originalMount.call(this)
      this.$nextTick(() => {
        const end = performance.now()
        this.$el.setAttribute('data-mount-time', (end - start).toFixed(2))
      })
    }
    
    // 响应式系统性能
    let reactiveAccessCount = 0
    const originalGet = Vue.reactive.__v_get
    Vue.reactive.__v_get = function(...args) {
      reactiveAccessCount++
      return originalGet.apply(this, args)
    }
    
    setInterval(() => {
      console.log(`⚡ 响应式访问次数: ${reactiveAccessCount}/秒`)
      reactiveAccessCount = 0
    }, 1000)
  }
  
  private recordMetric(name: string, value: number) {
    if (!this.metrics.has(name)) {
      this.metrics.set(name, [])
    }
    this.metrics.get(name)!.push(value)
  }
  
  getAverage(name: string): number {
    const values = this.metrics.get(name) || []
    return values.reduce((a, b) => a + b, 0) / values.length
  }
  
  getReport() {
    return {
      FCP: this.getAverage('FCP'),
      LCP: this.getAverage('LCP'),
      FID: this.getAverage('FID'),
      CLS: this.getAverage('CLS')
    }
  }
}

// 使用示例

monitor.measureFCP()
monitor.measureLCP()
monitor.measureFID()
monitor.measureCLS()
monitor.measureVueMetrics()

  },
  
  setup(props) {
    const scrollTop = ref(0)
    const startIndex = computed(() =>

      Math.min(
        startIndex.value + Math.ceil(props.containerHeight / props.itemHeight),

// 内存泄漏检测工具
class MemoryLeakDetector {
  private snapshots: Map<string, any[]> = new Map()
  
  takeSnapshot(name: string) {
    if (performance.memory) {
      this.snapshots.set(name, {
        used: performance.memory.usedJSHeapSize,
        total: performance.memory.totalJSHeapSize,
        limit: performance.memory.jsHeapSizeLimit
      })
    }
  }
  
  compareSnapshots(name1: string, name2: string) {
    const snap1 = this.snapshots.get(name1)
    const snap2 = this.snapshots.get(name2)
    
    if (!snap1 || !snap2) return null
    
    return {
      usedGrowth: snap2.used - snap1.used,
      totalGrowth: snap2.total - snap1.total,
      growthPercentage: ((snap2.used - snap1.used) / snap1.used * 100).toFixed(2)
    }
  }
  
  detectLeaks() {
    // 强制垃圾回收（开发环境）
    if (window.gc) {
      window.gc()
    }
    
    this.takeSnapshot('after-gc')
    
    // 检查DOM节点泄漏
    const nodes = document.querySelectorAll('*')
    console.log(`DOM节点数量: ${nodes.length}`)
  }
}

// 使用示例

detector.takeSnapshot('initial')

// 操作后
detector.takeSnapshot('after-operation')
const result = detector.compareSnapshots('initial', 'after-operation')
console.log('内存增长:', result)

import { onScopeDispose } from 'vue'

function useEventListener(target: EventTarget, event: string, callback: EventListener) {
  target.addEventListener(event, callback)
  
  // 自动清理
  onScopeDispose(() => {
    target.removeEventListener(event, callback)
  })
}

// 使用示例

    const buttonRef = ref<HTMLButtonElement>()
    
    onMounted(() => {
      useEventListener(buttonRef.value!, 'click', handleClick)
      useEventListener(window, 'resize', handleResize)
    })
    
    return { buttonRef }
  }
}

// 定时器清理
function useTimeout(callback: () => void, delay: number) {
  const timeout = setTimeout(callback, delay)
  
  onScopeDispose(() => {
    clearTimeout(timeout)
  })
  
  return () => clearTimeout(timeout)
}

function useInterval(callback: () => void, delay: number) {
  const interval = setInterval(callback, delay)
  
  onScopeDispose(() => {
    clearInterval(interval)
  })
  
  return () => clearInterval(interval)
}

// 高性能图片懒加载组件
const LazyImage = defineComponent({
  props: {
    src: String,
    alt: String,
    placeholder: String,
    loading: String
  },
  
  setup(props) {
    const isLoaded = ref(false)
    const isError = ref(false)
    const imgRef = ref<HTMLImageElement>()
    
    // 使用Intersection Observer
    const observer = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          loadImage()
          observer.disconnect()
        }
      })
    }, {
      rootMargin: '50px'
    })
    
    const loadImage = () => {
      const img = new Image()
      img.src = props.src!
      img.onload = () => {
        isLoaded.value = true
        if (imgRef.value) {
          imgRef.value.src = props.src!
        }
      }
      img.onerror = () => {
        isError.value = true
      }
    }
    
    onMounted(() => {
      if (imgRef.value) {
        observer.observe(imgRef.value)
      }
    })
    
    onUnmounted(() => {
      observer.disconnect()
    })
    
    return () => h('img', {
      ref: imgRef,
      src: isLoaded.value ? props.src : props.placeholder,
      alt: props.alt,
      loading: props.loading || 'lazy',
      class: {
        'lazy-image': true,
        'loaded': isLoaded.value,
        'error': isError.value
      }
    })
  }
})

### 2. 无限滚动优化

```typescript
// 高性能无限滚动
function useInfiniteScroll(
  loadMore: () => Promise<any[]>,
  options = {}
) {
  const {
    threshold = 100,
    maxPages = 10
  } = options
  
  const items = ref<any[]>([])
  const isLoading = ref(false)
  const hasMore = ref(true)
  const page = ref(1)
  
  const loadItems = async () => {
    if (isLoading.value || !hasMore.value) return
    
    isLoading.value = true
    try {
      const newItems = await loadMore(page.value)
      
      if (newItems.length === 0 || page.value >= maxPages) {
        hasMore.value = false
      } else {
        items.value.push(...newItems)
        page.value++
      }
    } finally {
      isLoading.value = false
    }
  }
  
  // 使用Intersection Observer
  const observer = new IntersectionObserver((entries) => {
    const lastEntry = entries[entries.length - 1]
    if (lastEntry.isIntersecting) {
      loadItems()
    }
  }, {
    rootMargin: `${threshold}px`
  })
  
  const observe = (element: Element) => {
    observer.observe(element)
  }
  
  const unobserve = (element: Element) => {
    observer.unobserve(element)
  }
  
  return {
    items,
    isLoading,
    hasMore,
    loadItems,
    observe,
    unobserve
  }
}

// 使用示例
export default {
  setup() {
    const loadProducts = async (page: number) => {
      const response = await api.getProducts({ page, limit: 20 })
      return response.data
    }
    
    const { items, isLoading, hasMore, observe } = useInfiniteScroll(loadProducts)
    
    return { products: items, isLoading, hasMore, observe }
  }
}
```

// 优化后：防抖+缓存+分页
const OptimizedSearchComponent = {
  setup() {
    const query = ref('')
    const results = ref([])
    const isLoading = ref(false)
    const cache = new Map()
    
    // 防抖搜索
    const debouncedSearch = useDebounce(async (searchQuery: string) => {
      if (cache.has(searchQuery)) {
        results.value = cache.get(searchQuery)
        return
      }
      
      isLoading.value = true
      try {
        const searchResults = await api.search(searchQuery)
        cache.set(searchQuery, searchResults)
        results.value = searchResults
      } finally {
        isLoading.value = false
      }
    }, 300)
    
    watch(query, debouncedSearch)
    
    return { query, results, isLoading }
  }
}
